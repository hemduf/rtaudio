if (WIN32)
    include_directories(../include)

    function(copy_runtime_deps target)
        if (WIN32)
            add_custom_command(
                TARGET ${target}
                POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    $<TARGET_RUNTIME_DLLS:${target}>
                    $<TARGET_FILE_DIR:${target}>
                COMMAND_EXPAND_LISTS
            )
        endif()
    endfunction()
endif (WIN32)
